cmake_minimum_required(VERSION 3.20)
project(OutRunReadbackHelpers LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(control_publish "${ROOT}/windows/host/control_publish.cpp")
target_include_directories(control_publish PRIVATE "${ROOT}/windows/host" "${ROOT}/windows/shim" "${ROOT}/windows/vendor/controls")
target_compile_definitions(control_publish PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000002)
add_executable(capture_replay "${ROOT}/windows/tests/capture_replay.cpp" "${ROOT}/windows/source/outrun_adapter.cpp" "${ROOT}/windows/shim/parse.cpp" "${ROOT}/windows/shim/core.cpp" "${ROOT}/src/input/dual_sense.cpp")
target_include_directories(capture_replay PRIVATE "${ROOT}/windows/shim" "${ROOT}/windows/host" "${ROOT}/windows/host/recorder" "${ROOT}/windows/source" "${ROOT}/windows/vendor/controls" "${ROOT}/windows/vendor/session" "${ROOT}/src")
target_link_libraries(capture_replay PRIVATE bcrypt)
foreach(t control_publish capture_replay)
  target_compile_features(${t} PRIVATE cxx_std_17)
  target_compile_options(${t} PRIVATE /W4 /WX /EHsc)
endforeach()
