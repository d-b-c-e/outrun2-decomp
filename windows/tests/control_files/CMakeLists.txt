cmake_minimum_required(VERSION 3.20)
project(shim_control_files CXX)
set(WIN ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(control_files main.cpp ${WIN}/shim/core.cpp ${WIN}/shim/parse.cpp)
target_include_directories(control_files PRIVATE ${WIN}/host ${WIN}/shim ${WIN}/vendor/controls ${WIN}/tests)
target_compile_features(control_files PRIVATE cxx_std_17)
target_compile_definitions(control_files PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000002)
target_compile_options(control_files PRIVATE /W4 /WX /EHsc)

add_executable(control_publish ${WIN}/host/control_publish.cpp)
target_include_directories(control_publish PRIVATE ${WIN}/host ${WIN}/shim ${WIN}/vendor/controls)
target_compile_features(control_publish PRIVATE cxx_std_17)
target_compile_definitions(control_publish PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000002)
target_compile_options(control_publish PRIVATE /W4 /WX /EHsc)
