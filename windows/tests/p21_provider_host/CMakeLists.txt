cmake_minimum_required(VERSION 3.20)
project(p21_provider_host CXX)
# P21k: the P21 host over zero's real Core and DInputProvider on fake COM (no device, game or launch). Also rebuilds
# zero's own provider tests against the extended fake (its gates are opt-in), as a regression check.
set(WIN ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(SHIM ${WIN}/shim/dinput_provider.cpp ${WIN}/shim/parse.cpp ${WIN}/shim/core.cpp)
foreach(t p21_provider_host test_dinput_provider)
    if(t STREQUAL p21_provider_host)
        add_executable(${t} main.cpp ${SHIM})
    else()
        add_executable(${t} ${WIN}/tests/test_dinput_provider.cpp ${SHIM})
    endif()
    target_include_directories(${t} PRIVATE ${WIN}/shim ${WIN}/tests ${WIN}/vendor/controls)
    target_compile_features(${t} PRIVATE cxx_std_17)
    target_compile_definitions(${t} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
    target_compile_options(${t} PRIVATE /W4 /EHsc)
    target_link_libraries(${t} PRIVATE dinput8 dxguid)
endforeach()
target_compile_options(p21_provider_host PRIVATE /WX)
