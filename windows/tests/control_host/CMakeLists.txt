cmake_minimum_required(VERSION 3.20)
project(shim_control_host CXX)
set(WIN ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(control_host main.cpp ${WIN}/shim/core.cpp ${WIN}/shim/parse.cpp ${WIN}/shim/dinput_provider.cpp)
target_include_directories(control_host PRIVATE ${WIN}/host ${WIN}/shim ${WIN}/vendor/controls ${WIN}/tests)
target_compile_features(control_host PRIVATE cxx_std_17)
target_compile_definitions(control_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(control_host PRIVATE /W4 /WX /EHsc)
target_link_libraries(control_host PRIVATE dinput8 dxguid)
