cmake_minimum_required(VERSION 3.20)
project(readonly_provider CXX)
set(WIN ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_executable(readonly_provider main.cpp ${WIN}/shim/core.cpp ${WIN}/shim/parse.cpp ${WIN}/shim/dinput_provider.cpp)
target_include_directories(readonly_provider PRIVATE ${WIN}/shim ${WIN}/tests ${WIN}/vendor/controls)
target_compile_features(readonly_provider PRIVATE cxx_std_17)
target_compile_definitions(readonly_provider PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
target_compile_options(readonly_provider PRIVATE /W4 /WX /EHsc)
target_link_libraries(readonly_provider PRIVATE dinput8 dxguid user32)
add_executable(test_core_reports ${WIN}/tests/test_shim.cpp ${WIN}/shim/core.cpp ${WIN}/shim/parse.cpp)
target_include_directories(test_core_reports PRIVATE ${WIN}/shim ${WIN}/tests ${WIN}/vendor/controls)
target_compile_features(test_core_reports PRIVATE cxx_std_17)
target_compile_options(test_core_reports PRIVATE /W4 /WX /EHsc)
add_executable(wgi_inventory wgi_inventory.cpp)
target_compile_features(wgi_inventory PRIVATE cxx_std_20)
target_compile_definitions(wgi_inventory PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(wgi_inventory PRIVATE /W4 /WX /EHsc)
target_link_libraries(wgi_inventory PRIVATE windowsapp)
add_executable(dinput_inventory dinput_inventory.cpp)
target_compile_features(dinput_inventory PRIVATE cxx_std_17)
target_compile_definitions(dinput_inventory PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
target_compile_options(dinput_inventory PRIVATE /W4 /WX /EHsc)
target_link_libraries(dinput_inventory PRIVATE dinput8 dxguid)
