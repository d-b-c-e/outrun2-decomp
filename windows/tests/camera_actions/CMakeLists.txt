cmake_minimum_required(VERSION 3.20)
project(camera_actions_compat CXX)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(CONTROLS ${ROOT}/../controls)
set(SESSION ${ROOT}/../session)
add_library(shim_facade OBJECT ${ROOT}/shim_dll/dbce_shim.cpp ${ROOT}/shim/core.cpp ${ROOT}/shim/parse.cpp ${ROOT}/shim/dinput_provider.cpp ${ROOT}/shim/bridge_provider.cpp)
target_include_directories(shim_facade PUBLIC ${ROOT}/shim ${ROOT}/host ${ROOT}/host/recorder ${ROOT}/shim_dll ${ROOT}/tests ${CONTROLS} ${SESSION})
target_compile_features(shim_facade PUBLIC cxx_std_17)
target_compile_definitions(shim_facade PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE DBCE_SHIM_BUILD)
target_compile_options(shim_facade PUBLIC /W4 /WX /EHsc)
add_library(dbce_shim SHARED $<TARGET_OBJECTS:shim_facade>)
target_link_libraries(dbce_shim PRIVATE dinput8 dxguid bcrypt)
add_executable(shim_abi2 ${ROOT}/tests/test_shim_abi.cpp)
target_link_libraries(shim_abi2 PRIVATE shim_facade dinput8 dxguid bcrypt)
add_executable(shim_core ${ROOT}/tests/test_shim.cpp ${ROOT}/shim/core.cpp ${ROOT}/shim/parse.cpp)
target_include_directories(shim_core PRIVATE ${ROOT}/shim ${ROOT}/tests ${CONTROLS})
target_compile_features(shim_core PRIVATE cxx_std_17)
target_compile_definitions(shim_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(shim_core PRIVATE /W4 /EHsc)
enable_testing()
add_test(NAME shim_abi2 COMMAND shim_abi2 $<TARGET_FILE:dbce_shim>)
add_test(NAME shim_core COMMAND shim_core)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_executable(camera_actions_compat main.cpp ${ROOT}/shim/parse.cpp ${ROOT}/shim/core.cpp)
    target_include_directories(camera_actions_compat PRIVATE ${ROOT}/shim ${ROOT}/host/recorder ${ROOT}/shim_dll ${ROOT}/tests ${CONTROLS} ${SESSION})
    target_compile_features(camera_actions_compat PRIVATE cxx_std_17)
    target_compile_definitions(camera_actions_compat PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_compile_options(camera_actions_compat PRIVATE /W4 /WX /EHsc)
    add_test(NAME camera_actions_compat COMMAND camera_actions_compat ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)
endif()
